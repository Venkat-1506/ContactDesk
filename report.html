Aim

To design and develop a responsive, component-based single-page React personal network explorer and contact management platform (ContactDesk / People Atlas) that delivers real-time contact management, live multi-attribute searching, alphabetical A-Z cataloging, spatial geographical place grouping, close-circle bookmarking, interaction timeline tracking, algorithmic data health and network balance scoring, interactive SVG and Python Matplotlib visual chart analytics, JSON backup export/import capabilities, and browser LocalStorage data persistence.

Procedure

  1. Project Architecture & Structure
    • Single-page application (SPA) built with React 19, React Router v7, Vite, and JavaScript ES6+.
    • App.jsx acts as the primary layout and router shell managing navigation across product views: Network Insights (/insights), People Directory (/people), Places Explorer (/places), Recent Activity (/recent), Favorites (/favorites), Data Management (/data), Contact Details (/contact/:id), Add Person (/add-contact), and Edit Person (/edit-contact/:id).
    • Organized into a modular architecture separating reusable UI components, page views, storage utilities, calculation engines, sample data presets, and styling systems.
  2. Data Services & Local Persistence Integration
    • contactStorage.js handles data persistence via browser LocalStorage API (contactdesk_contacts), guaranteeing 100% client-side privacy without external server dependencies.
    • Automatic data normalization (normalizeContact) ensuring field integrity across unique IDs, names, phones, emails, categories, relationships, locations, organizations, timestamps, and favorite flags.
    • Robust backup system supporting full JSON export (exportContactsJSON), safe deduplicated import (importContactsJSON), initial sample data loading (loadSampleContacts), and complete storage resets.
  3. State Management & Analytics Engines
    • insightCalculations.js executes live algorithmic calculations on contact records for category distributions, location stats, relationship mixes, close circle ratios, field completeness percentages, automated standout observations, and network balance metrics.
    • generate_insights_charts.py provides a secondary Python Matplotlib visualization pipeline generating static chart PNG figures (group_distribution.png, network_composition.png, top_places.png, data_health.png).
  4. UI Components & Views
    • Atlas.jsx (Insights view) renders editorial metric summary cards, dynamic bar charts, and interactive SVG donut charts (InsightsDonutChart.jsx) with live hover legends.
    • People.jsx & ContactList.jsx provide multi-attribute live search with keyboard shortcuts (/), alphabetical Monogram sectioning (Monogram.jsx), and touch-optimized action links (tel: and mailto:).
    • Places.jsx groups contact records spatially by city/location; Recent.jsx tracks activity timelines and offers one-click "Mark as Contacted Today" interactions.
    • ContactForm.jsx & ContactDetails.jsx offer full record management with safety confirmation modals (ConfirmDialog.jsx) and toast notifications (Toast.jsx).
  5. Styling & Responsive Editorial Design
    • Pure Vanilla CSS (App.css & index.css) with custom CSS properties driving a paper-texture archival design system (--bg-texture), ink typography (Bricolage Grotesque, IBM Plex Sans, IBM Plex Mono), and category color badges.
    • Fully responsive layout featuring an archival top header (Navbar.jsx) with sticky navigation, quick search triggers, and one-touch action buttons across mobile, tablet, and desktop viewports.

import { initialContacts } from '../data/initialContacts';

const STORAGE_KEY = 'contactdesk_contacts';
const ONBOARDED_KEY = 'contactdesk_onboarded';

export function normalizeContact(contact) {
  if (!contact || typeof contact !== 'object') return null;
  const now = new Date().toISOString();
  return {
    id: contact.id || (Date.now().toString() + Math.random().toString(36).substring(2, 6)),
    name: contact.name ? contact.name.trim() : 'Unnamed Person',
    phone: contact.phone ? contact.phone.trim() : '',
    email: contact.email ? contact.email.trim() : '',
    category: contact.category || 'Other',
    relationship: contact.relationship || contact.category || 'Other',
    organization: contact.organization || '',
    location: contact.location || '',
    notes: contact.notes || '',
    favorite: Boolean(contact.favorite),
    createdAt: contact.createdAt || now,
    updatedAt: contact.updatedAt || now,
    lastContactedAt: contact.lastContactedAt || null
  };
}

export function getContacts() {
  try {
    const data = localStorage.getItem(STORAGE_KEY);
    if (!data) {
      return loadSampleContacts();
    }
    const parsed = JSON.parse(data);
    if (!Array.isArray(parsed) || parsed.length === 0) return [];
    return parsed.map(normalizeContact);
  } catch (error) {
    console.error('Malformed localStorage data detected, clearing safely:', error);
    saveContacts([]);
    return [];
  }
}

export function saveContacts(contacts) {
  try {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(contacts));
  } catch (error) {
    console.error('Failed to save contacts to localStorage:', error);
  }
}

export function addContact(contactData) {
  const contacts = getContacts();
  const now = new Date().toISOString();
  const newContact = normalizeContact({
    ...contactData,
    createdAt: now,
    updatedAt: now
  });
  const updatedList = [newContact, ...contacts];
  saveContacts(updatedList);
  return newContact;
}

export function updateContact(id, updatedFields) {
  const contacts = getContacts();
  let updatedContact = null;
  const updatedList = contacts.map((contact) => {
    if (contact.id === id) {
      updatedContact = normalizeContact({
        ...contact,
        ...updatedFields,
        updatedAt: new Date().toISOString()
      });
      return updatedContact;
    }
    return contact;
  });
  saveContacts(updatedList);
  return updatedContact;
}

export function toggleFavorite(id) {
  const contacts = getContacts();
  let toggledState = false;
  const updatedList = contacts.map((c) => {
    if (c.id === id) {
      toggledState = !c.favorite;
      return {
        ...c,
        favorite: toggledState,
        updatedAt: new Date().toISOString()
      };
    }
    return c;
  });
  saveContacts(updatedList);
  return toggledState;
}

export function exportContactsJSON() {
  const contacts = getContacts();
  return JSON.stringify(contacts, null, 2);
}

src/components/InsightsDonutChart.jsx

import React, { useState } from 'react';

export default function InsightsDonutChart({ groupStats = [] }) {
  const [hoveredSlice, setHoveredSlice] = useState(null);

  if (!groupStats || groupStats.length === 0) return null;

  const R = 40;
  const C = 2 * Math.PI * R;
  let cumulativeOffset = 0;
  const totalCount = groupStats.reduce((acc, curr) => acc + curr.count, 0);

  const slices = groupStats.map((item) => {
    const fraction = totalCount > 0 ? item.count / totalCount : 0;
    const sliceLength = fraction * C;
    const offset = cumulativeOffset;
    cumulativeOffset += sliceLength;
    return { ...item, fraction, sliceLength, offset };
  });

  return (
    <div className="insights-donut-container">
      <div className="insights-donut-graphic">
        <svg viewBox="0 0 120 120" className="insights-donut-svg">
          {slices.map((slice) => {
            const isHovered = hoveredSlice?.category === slice.category;
            return (
              <circle
                key={slice.category}
                cx="60"
                cy="60"
                r={R}
                fill="transparent"
                stroke={slice.color}
                strokeWidth={isHovered ? '18' : '14'}
                strokeDasharray={`${slice.sliceLength} ${C - slice.sliceLength}`}
                strokeDashoffset={-slice.offset}
                transform="rotate(-90 60 60)"
                onMouseEnter={() => setHoveredSlice(slice)}
                onMouseLeave={() => setHoveredSlice(null)}
              />
            );
          })}
          <text x="60" y="56" textAnchor="middle" className="donut-center-num mono-font">
            {hoveredSlice ? hoveredSlice.count : totalCount}
          </text>
          <text x="60" y="70" textAnchor="middle" className="donut-center-label mono-font">
            {hoveredSlice ? hoveredSlice.category.toUpperCase() : 'CONTACTS'}
          </text>
        </svg>
      </div>
    </div>
  );
}

import React, { useState, useEffect, useMemo } from 'react';
import { Link } from 'react-router-dom';
import Monogram from '../components/Monogram';
import Icon from '../components/Icon';
import { getContacts } from '../utils/contactStorage';

export default function Places() {
  const [contacts, setContacts] = useState([]);
  const [selectedPlace, setSelectedPlace] = useState('ALL');

  useEffect(() => {
    setContacts(getContacts());
  }, []);

  const locationGroups = useMemo(() => {
    const map = new Map();
    contacts.forEach((c) => {
      const loc = c.location && c.location.trim() ? c.location.trim() : 'Unspecified Location';
      if (!map.has(loc)) map.set(loc, []);
      map.get(loc).push(c);
    });
    return Array.from(map.entries())
      .map(([place, people]) => ({ place, people, count: people.length }))
      .sort((a, b) => b.count - a.count);
  }, [contacts]);

  return (
    <div className="page-container places-page-layout">
      <header className="page-editorial-header">
        <div className="header-titles">
          <h1 className="page-title">PLACES</h1>
          <p className="page-subtitle">Explore the people in your network by where they are.</p>
        </div>
      </header>

      <div className="places-list-stack">
        {locationGroups.map((group) => (
          <div key={group.place} className="place-group-block">
            <div className="place-header-rule">
              <h3 className="place-heading">{group.place.toUpperCase()}</h3>
              <span className="count-tag">{group.count} PEOPLE</span>
            </div>
          </div>
        ))}
      </div>
    </div>
  );
}
